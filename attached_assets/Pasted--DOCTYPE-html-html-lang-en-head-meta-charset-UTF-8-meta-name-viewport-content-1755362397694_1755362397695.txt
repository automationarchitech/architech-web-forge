    <script>
        const speechText = "I need to write an email to my boss Sarah about Project Blue Agent and explain what's going on with the database migration delays and our plan to get back on track.";
        const words = speechText.split(' ');
        let currentWordIndex = 0;
        let speechInterval;
        let isPlaying = false;

        function resetDemo() {
            document.getElementById('speechText').innerHTML = '';
            
            // Reset function call 1
            const functionCall1 = document.getElementById('functionCall');
            functionCall1.style.display = 'none';
            functionCall1.style.background = '#f093fb';
            functionCall1.innerHTML = '<div class="function-icon">⚡</div><span>get_contact_info("Sarah") executing...</span>';
            
            // Reset function call 2
            const functionCall2 = document.getElementById('functionCall2');
            functionCall2.style.display = 'none';
            functionCall2.style.background = '#f093fb';
            functionCall2.innerHTML = '<div class="function-icon">⚡</div><span>fetch_project("Blue Agent") executing...</span>';
            
            document.getElementById('typingIndicator').style.display = 'none';
            document.getElementById('aiResponse').style.display = 'none';
            document.getElementById('demoControls').style.display = 'none';  // Hide replay button
            currentWordIndex = 0;
            isPlaying = false;
            if (speechInterval) clearInterval(speechInterval);
        }

        function startDemo() {
            if (isPlaying) return;
            isPlaying = true;
            resetDemo();
            
            const speechTextElement = document.getElementById('speechText');
            
            speechInterval = setInterval(() => {
                if (currentWordIndex < words.length) {
                    const wordSpan = document.createElement('span');
                    wordSpan.className = 'word';
                    wordSpan.textContent = words[currentWordIndex] + ' ';
                    wordSpan.style.animationDelay = '0ms';
                    speechTextElement.appendChild(wordSpan);
                    
                    // Trigger contact function after "my boss Sarah"
                    if (currentWordIndex === 9) { // After "Sarah"
                        setTimeout(() => {
                            document.getElementById('functionCall').style.display = 'flex';
                        }, 200);
                        
                        // Complete contact function before user finishes
                        setTimeout(() => {
                            document.getElementById('functionCall').innerHTML = '<div class="function-icon">✅</div><span>get_contact_info() completed</span>';
                            document.getElementById('functionCall').style.background = '#68d391';
                        }, 1800);
                    }
                    
                    // Trigger project function after "Project Blue Agent"
                    if (currentWordIndex === 13) { // After "Agent"
                        setTimeout(() => {
                            document.getElementById('functionCall2').style.display = 'flex';
                        }, 200);
                        
                        // Complete project function before user finishes
                        setTimeout(() => {
                            document.getElementById('functionCall2').innerHTML = '<div class="function-icon">✅</div><span>fetch_project("Blue Agent") completed</span>';
                            document.getElementById('functionCall2').style.background = '#68d391';
                        }, 2200);
                    }
                    
                    currentWordIndex++;
                } else {
                    clearInterval(speechInterval);
                    // Show typing indicator after speech completes
                    setTimeout(() => {
                        document.getElementById('functionCall').style.display = 'none';
                        document.getElementById('functionCall2').style.display = 'none';
                        document.getElementById('typingIndicator').style.display = 'flex';
                    }, 800);
                    
                    // Show final response
                    setTimeout(() => {
                        document.getElementById('typingIndicator').style.display = 'none';
                        document.getElementById('aiResponse').style.display = 'block';
                        document.getElementById('demoControls').style.display = 'block';  // Show replay button
                        isPlaying = false;
                    }, 2000);
                }
            }, 400); // Adjust timing as needed
        }

        // Auto-start demo when page loads
        window.addEventListener('load', () => {
            setTimeout(startDemo, 1000);
        });
    </script>
